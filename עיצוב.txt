<!-- SCRIPT FOR TYPEWRITER EFFECT, SCROLL TRACKING, AND RESPONSIVE INTERACTIVITY -->
<script>
    // Initialize Lucide icons
    lucide.createIcons();

    // DOM Elements
    const chatWindow = document.getElementById('chat-window');
    const messagesContainer = document.getElementById('messages-container');
    const scrollToBottomBtn = document.getElementById('scroll-to-bottom-btn');
    const scrollBottomAnchor = document.getElementById('scroll-bottom-anchor');
    const typewriterEl = document.getElementById('typewriter-text');
    const sourceBadge = document.getElementById('source-badge');
    const userInput = document.getElementById('user-input');
    const mobileMenuBtn = document.getElementById('mobile-menu-btn');
    const mobileMenu = document.getElementById('mobile-menu');

    // Mobile nav menu toggle
    if (mobileMenuBtn && mobileMenu) {
      mobileMenuBtn.addEventListener('click', () => {
        mobileMenu.classList.toggle('hidden');
      });
    }

    // The authentic response text to type out smoothly character-by-character
    const fullAnswerText = `במקרה של נזילת מים או פיצוץ צנרת בדירה המבוטחת, ניתן להגיש תביעה דיגיטלית ב-3 שלבים:

1. צילום מוקד הנזק: יש לתעד את מקור הדליפה, הרטיבות בקירות והנזק לתכולה לפני תחילת עבודות.
2. בחירת שרברב: 
   • שרברב שבהסדר הפניקס: השתתפות עצמית מופחתת והתחשבנות ישירה ללא צורך בהוצאת כספים מראש.
   • שרברב פרטי לבחירתך: נדרשת הגשת חשבונית מס ודוח טכני מפורט.
3. פתיחת תביעה מקוונת: יש להגיש את הטופס הדיגיטלי בפורטל התביעות (homeclaim) בצירוף התמונות.

באתר מצוין כי מוקד החירום הטלפוני לנזקי צנרת פעיל 24/7 במספר 5040*.`;

    // Typewriter state
    let charIndex = 0;
    let isTyping = true;
    let typingSpeed = 16; // milliseconds per character

    function typeNextChar() {
      if (charIndex < fullAnswerText.length) {
        const char = fullAnswerText.charAt(charIndex);
        if (char === '\n') {
          typewriterEl.innerHTML += '<br>';
        } else {
          typewriterEl.innerHTML += char;
        }
        charIndex++;
        // Maintain scroll at bottom during typing
        messagesContainer.scrollTop = messagesContainer.scrollHeight;
        setTimeout(typeNextChar, typingSpeed);
      } else {
        // Typing finished
        isTyping = false;
        typewriterEl.classList.remove('cursor-blink');
        sourceBadge.classList.remove('hidden');
        lucide.createIcons();
        messagesContainer.scrollTop = messagesContainer.scrollHeight;
      }
    }

    // Start typewriter effect automatically on load
    window.addEventListener('DOMContentLoaded', () => {
      setTimeout(typeNextChar, 500);
    });

    // Detect user scrolling away from bottom to show the "Scroll to Bottom" button
    messagesContainer.addEventListener('scroll', () => {
      const scrollPos = messagesContainer.scrollTop + messagesContainer.clientHeight;
      const totalHeight = messagesContainer.scrollHeight;
      const distanceFromBottom = totalHeight - scrollPos;

      if (distanceFromBottom > 70) {
        scrollToBottomBtn.classList.remove('opacity-0', 'pointer-events-none', 'translate-y-2');
        scrollToBottomBtn.classList.add('opacity-100', 'translate-y-0');
      } else {
        scrollToBottomBtn.classList.add('opacity-0', 'pointer-events-none', 'translate-y-2');
        scrollToBottomBtn.classList.remove('opacity-100', 'translate-y-0');
      }
    });

    // Smooth Scroll to bottom
    function scrollToBottomSmooth() {
      messagesContainer.scrollTo({
        top: messagesContainer.scrollHeight,
        behavior: 'smooth'
      });
    }

    // Toggle Chat Window minimize/open
    let isChatOpen = true;
    function toggleChatWindow() {
      isChatOpen = !isChatOpen;
      const chevron = document.getElementById('launcher-chevron');
      if (isChatOpen) {
        chatWindow.classList.remove('hidden', 'scale-90', 'opacity-0');
        chatWindow.classList.add('scale-100', 'opacity-100');
        if (chevron) chevron.setAttribute('data-lucide', 'chevron-down');
      } else {
        chatWindow.classList.add('scale-90', 'opacity-0');
        setTimeout(() => {
          chatWindow.classList.add('hidden');
        }, 200);
        if (chevron) chevron.setAttribute('data-lucide', 'chevron-up');
      }
      lucide.createIcons();
    }

    document.getElementById('btn-minimize').addEventListener('click', toggleChatWindow);

    document.getElementById('btn-reset').addEventListener('click', () => {
      typewriterEl.innerHTML = '';
      typewriterEl.classList.add('cursor-blink');
      sourceBadge.classList.add('hidden');
      charIndex = 0;
      isTyping = true;
      typeNextChar();
    });

    // Sample question pill click
    function handleSampleQuery(queryText) {
      userInput.value = queryText;
      submitUserMessage();
    }

    // Handle user submitting message
    function submitUserMessage() {
      const text = userInput.value.trim();
      if (!text) return;

      // Append user bubble
      const userBubble = document.createElement('div');
      userBubble.className = "flex items-start justify-end gap-2.5";
      userBubble.innerHTML = `
        <div class="max-w-[85%] bg-[#0c172d] text-white rounded-2xl rounded-tl-none p-3.5 shadow-sm text-sm leading-relaxed">
          ${escapeHtml(text)}
        </div>
        <div class="w-7 h-7 rounded-lg bg-slate-300 text-slate-700 flex items-center justify-center shrink-0 shadow-sm mt-0.5">
          <i data-lucide="user" class="w-4 h-4"></i>
        </div>
      `;
      messagesContainer.insertBefore(userBubble, scrollBottomAnchor);

      userInput.value = '';
      lucide.createIcons();
      scrollToBottomSmooth();

      // Show typing indicator bubble
      const typingIndicator = document.createElement('div');
      typingIndicator.id = "temp-indicator";
      typingIndicator.className = "flex items-start gap-2.5";
      typingIndicator.innerHTML = `
        <div class="w-7 h-7 rounded-lg bg-[#ff551f] text-white flex items-center justify-center shrink-0 shadow-sm mt-0.5">
          <i data-lucide="bot" class="w-4 h-4"></i>
        </div>
        <div class="bg-white border border-slate-200 rounded-2xl rounded-tr-none px-4 py-3 shadow-bubble flex items-center gap-1.5 text-xs text-slate-500">
          <span class="w-2 h-2 rounded-full bg-[#ff551f] animate-pulse"></span>
          <span>מחפש ב-1,091 עמודי האתר ומפעיל RAG היברידי...</span>
        </div>
      `;
      messagesContainer.insertBefore(typingIndicator, scrollBottomAnchor);
      lucide.createIcons();
      scrollToBottomSmooth();

      // Simulate streaming response
      setTimeout(() => {
        const ind = document.getElementById('temp-indicator');
        if (ind) ind.remove();

        const newBotBubble = document.createElement('div');
        newBotBubble.className = "flex items-start gap-2.5";
        const uniqueId = 'stream-' + Date.now();
        newBotBubble.innerHTML = `
          <div class="w-7 h-7 rounded-lg bg-[#ff551f] text-white flex items-center justify-center shrink-0 shadow-sm mt-0.5">
            <i data-lucide="bot" class="w-4 h-4"></i>
          </div>
          <div class="max-w-[88%] bg-white border border-slate-200 rounded-2xl rounded-tr-none p-3.5 shadow-bubble text-slate-800 text-sm leading-relaxed">
            <div id="${uniqueId}" class="cursor-blink text-slate-800 text-xs leading-relaxed space-y-2 mb-2"></div>
            <div class="pt-2 border-t border-slate-100 flex items-center justify-between text-[11px] text-slate-500">
              <span class="text-emerald-600 font-semibold flex items-center gap-1">
                <i data-lucide="check-circle" class="w-3.5 h-3.5"></i>
                מקור: fnx.co.il/service
              </span>
              <a href="#" class="text-[#ff551f] hover:underline">קרא עוד באתר</a>
            </div>
          </div>
        `;
        messagesContainer.insertBefore(newBotBubble, scrollBottomAnchor);
        lucide.createIcons();

        const responseSample = `לפי המידע העדכני באתר הפניקס בנושא זה, השירות ניתן ישירות בדיגיטל דרך אפליקציית הפניקס או בפורטל השירות. במידה ומדובר בבקשה לפוליסה ספציפית, ניתן להתחבר לאזור האישי באמצעות קוד חד-פעמי (SMS).`;
        let sIdx = 0;
        const targetDiv = document.getElementById(uniqueId);
        function typeStream() {
          if (sIdx < responseSample.length) {
            targetDiv.innerHTML += responseSample.charAt(sIdx);
            sIdx++;
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
            setTimeout(typeStream, 15);
          } else {
            targetDiv.classList.remove('cursor-blink');
          }
        }
        typeStream();

      }, 900);
    }

    function escapeHtml(text) {
      const map = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };
      return text.replace(/[&<>"']/g, function(m) { return map[m]; });
    }
  </script>
